-- first page
SELECT id, created_at, total FROM orders WHERE customer_id = :c ORDER BY created_at DESC, id DESC LIMIT
-- next page: cursor = (created_at, id) of the last row; tie-breaker column makes the order total
SELECT id, created_at, total FROM orders
WHERE customer_id = :c AND (created_at, id) < (:last_created_at, :last_id)
ORDER BY created_at DESC, id DESC LIMIT 20;
;
