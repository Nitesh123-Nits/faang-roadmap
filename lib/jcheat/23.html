@Service
public class TransferService {
    @Transactional(isolation = Isolation.READ_COMMITTED, timeout = 5)
    public void transfer(UUID from, UUID to, BigDecimal amount) {
        Account a = accounts.findByIdForUpdate(from).orElseThrow();
        Account b = accounts.findByIdForUpdate(to).orElseThrow();
        a.debit(amount); b.credit(amount);
        outbox.save(OutboxEvent.of("TransferCompleted", from, to, amount)); // same TX (see outbox pattern)
    }

    @Transactional(readOnly = true)
    public AccountView view(UUID id) { return accounts.findViewById(id).orElseThrow(); }
}
