Templates worth writing from memory

Thirteen skeletons that cover most interview problems. Each is compiled and tested against a brute-force oracle. For graph algorithms and DSU see the Toolkit (Chapter M3) and Chapters 52 and 53; for tries see Chapter 94.

import java.util.*;
import java.util.function.IntPredicate;

/** Interview-day templates: the skeletons worth writing from memory. */
public class Templates {

    /** First index i with a[i] >= target (a.length if none). Half-open [lo, hi). */
    public static int lowerBound(int[] a, int target) {
        int lo = 0, hi = a.length;
        while (lo < hi) {
            int mid = lo + (hi - lo) / 2;
            if (a[mid] >= target) hi = mid;
            else lo = mid + 1;
        }
        return lo;
    }

    /** Smallest x in [lo, hi] with ok(x) true, or hi + 1. ok must be monotone. */
    public static int firstTrue(int lo, int hi, IntPredicate ok) {
        int l = lo, r = hi + 1;
        while (l < r) {
            int mid = l + (r - l) / 2;
            if (ok.test(mid)) r = mid;
            else l = mid + 1;
        }
        return l;
    }

    /** Two pointers on a sorted array: indices of a pair summing to target. */
    public static int[] twoSumSorted(int[] a, int target) {
        int i = 0, j = a.length - 1;
        while (i < j) {
            long sum = (long) a[i] + a[j];
            if (sum == target) return new int[]{i, j};
            if (sum < target) i++;
            else j--;
        }
        return new int[]{-1, -1};
    }

    /** Variable sliding window: longest substring without a repeated character. */
    public static int longestUniqueWindow(String s) {
        int[] last = new int[128];          // last index + 1 of each character
        int best = 0, left = 0;
        for (int right = 0; right < s.length(); right++) {
            char c = s.charAt(right);
            left = Math.max(left, last[c]);
            last[c] = right + 1;
            best = Math.max(best, right - left + 1);
        }
        return best;
    }

    /** Prefix sum + hash map: number of subarrays whose sum equals k. */
    public static int subarraySumEqualsK(int[] a, int k) {
        Map<Long, Integer> seen = new HashMap<>();
        seen.put(0L, 1);
        long prefix = 0;
        int count = 0;
        for (int x : a) {
            prefix += x;
            count += seen.getOrDefault(prefix - k, 0);

        return count;

    /** Monotonic stack: next strictly greater value to the right, or -1. */
    public static int[] nextGreater(int[] a) {
        int[] out = new int[a.length];
        Arrays.fill(out, -1);
        Deque<Integer> stack = new ArrayDeque<>();   // indices, values decreasing
        for (int i = 0; i < a.length; i++) {
            while (!stack.isEmpty() && a[stack.peek()] < a[i])
                out[stack.pop()] = a[i];
            stack.push(i);
        }
        return out;
    }

    /** BFS on a grid (0 open, 1 wall): steps top-left to bottom-right, or -1. */
    public static int shortestPathGrid(int[][] grid) {
        int rows = grid.length, cols = grid[0].length;
        if (grid[0][0] == 1 || grid[rows - 1][cols - 1] == 1) return -1;
        int[][] dist = new int[rows][cols];
        for (int[] row : dist) Arrays.fill(row, -1);
        int[][] dirs = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
        Deque<int[]> queue = new ArrayDeque<>();
        dist[0][0] = 0;
        queue.offer(new int[]{0, 0});
        while (!queue.isEmpty()) {
            int[] cur = queue.poll();
            for (int[] d : dirs) {
                int r = cur[0] + d[0], c = cur[1] + d[1];
                if (r < 0 || r >= rows || c < 0 || c >= cols) continue;
                if (grid[r][c] == 1 || dist[r][c] != -1) continue;
                dist[r][c] = dist[cur[0]][cur[1]] + 1;
                queue.offer(new int[]{r, c});
            }
        }
        return dist[rows - 1][cols - 1];
    }

    /** Kahn's algorithm: a topological order, or empty if there is a cycle. */
    public static int[] kahnOrder(int n, int[][] edges) {
        List<List<Integer>> g = new ArrayList<>();
        for (int i = 0; i < n; i++) g.add(new ArrayList<>());
        int[] indeg = new int[n];
        for (int[] e : edges) {
            g.get(e[0]).add(e[1]);
            indeg[e[1]]++;
        }
        Deque<Integer> queue = new ArrayDeque<>();
        for (int i = 0; i < n; i++) if (indeg[i] == 0) queue.offer(i);
        int[] order = new int[n];
        int k = 0;
        while (!queue.isEmpty()) {
            int u = queue.poll();
            order[k++] = u;
            for (int v : g.get(u)) if (--indeg[v] == 0) queue.offer(v);
        }
        return k == n ? order : new int[0];
    }

    /** Backtracking: choose, explore, unchoose. All subsets of distinct values. */
    public static List<List<Integer>> subsets(int[] nums) {
        List<List<Integer>> out = new ArrayList<>();
        explore(nums, 0, new ArrayList<>(), out);
        return out;
    }

    private static void explore(int[] nums, int start, List<Integer> path,
                                List<List<Integer>> out) {
        out.add(new ArrayList<>(path));                 // copy: path keeps changing
        for (int i = start; i < nums.length; i++) {
            path.add(nums[i]);                          // choose
            explore(nums, i + 1, path, out);            // explore
            path.remove(path.size() - 1);               // unchoose
        }
    }

    /** 0/1 knapsack, one row: capacity runs downward so each item is used once. */
    public static int knapsack01(int[] weight, int[] value, int capacity) {
        int[] dp = new int[capacity + 1];
        for (int i = 0; i < weight.length; i++)
            for (int c = capacity; c >= weight[i]; c--)
                dp[c] = Math.max(dp[c], dp[c - weight[i]] + value[i]);
        return dp[capacity];
    }

    /** Unbounded coin change: fewest coins for amount, or -1. */
    public static int coinChange(int[] coins, int amount) {
        int inf = amount + 1;
        int[] dp = new int[amount + 1];
        Arrays.fill(dp, inf);
        dp[0] = 0;
        for (int c = 1; c <= amount; c++)
            for (int coin : coins)
                if (coin <= c) dp[c] = Math.min(dp[c], dp[c - coin] + 1);
        return dp[amount] >= inf ? -1 : dp[amount];
    }

    /** Longest strictly increasing subsequence in O(n log n) with tails[]. */
    public static int lisLength(int[] a) {
        int[] tails = new int[a.length];
        int size = 0;
        for (int x : a) {
            int lo = 0, hi = size;
            while (lo < hi) {
                int mid = lo + (hi - lo) / 2;
                if (tails[mid] < x) lo = mid + 1;
                else hi = mid;
            }
            tails[lo] = x;
            if (lo == size) size++;
        }
        return size;
    }

    /** Iterative inorder traversal with an explicit stack. */
    public static List<Integer> inorder(TreeNode root) {
        List<Integer> out = new ArrayList<>();
        Deque<TreeNode> stack = new ArrayDeque<>();
        TreeNode cur = root;
        while (cur != null || !stack.isEmpty()) {
            while (cur != null) {
                stack.push(cur);
                cur = cur.left;
            }
            cur = stack.pop();
            out.add(cur.val);
            cur = cur.right;
        }
        return out;
    }
}

FOUR JAVA TRAPS

Do not modify a collection while iterating it with a for-each loop (ConcurrentModificationException); do not compare boxed Integer with ==; do not use Arrays.asList on an int[] (it creates a one-element list of the array); do not forget to flush() a buffered writer.
