public void backup(Device d) {
  if (d != null
      && d.getStatus() == DeviceStatus.ONLINE
      && d.getIp() != null) {
    String cfg = ssh.run(d.getIp(),
        "show running-config");
    if (cfg != null && !cfg.isBlank()) {
      repo.save(new Backup(d.getId(), cfg,
          Instant.now()));
      log.info("Backup done for {}", d.getId());
    }
  }
}

public void backup(Device device) {
  if (!isBackupEligible(device)) return;

  String config = fetchRunningConfig(device);
  if (config.isBlank()) return;

  saveBackup(device, config);
}

private boolean isBackupEligible(Device d) {
  return d != null
      && d.getStatus() == DeviceStatus.ONLINE
      && d.getIp() != null;
}
