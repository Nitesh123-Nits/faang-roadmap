Spring Boot application.yml

Grouped by prefix, kebab-case, units on values:

order-service:
  retry:
    max-attempts: 3
    backoff-delay: 500ms
  kafka:
    topics:
      order-confirmed: orders.confirmed
spring:
  datasource:
    password: ${DB_PASSWORD}

It binds to a typed properties class:

@ConfigurationProperties(prefix = "order-service")
public record OrderServiceProperties(Retry retry) {
  public record Retry(int maxAttempts,
      Duration backoffDelay) {}
}
